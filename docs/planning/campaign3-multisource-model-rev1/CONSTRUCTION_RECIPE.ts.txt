export function multisourceBase(){

 const contentA=rec(content.find(v=>key(f(rec(v,170n),1n))===key(msId(1038,'content/task-a')))!,170n),b=msId(1027,'definition/task-b');
 const contentB=replace(contentA,{1:msId(1038,'content/task-b'),3:list([b]),7:list([b]),8:list([b]),10:list([b]),11:list([b]),13:list([b]),16:list([b])});
 const bodyCriterion=replace(rec(rows.find(v=>typeof v!=='boolean'&&v.kind==='record'&&v.schema.typeId===171n&&key(f(v,1n))===key(msId(1027,'definition/embodied-pressure')))!,171n),{1:msDefinition('body-criterion')});

  if(typeof definition!=='boolean'&&definition.kind==='record'&&definition.schema.typeId===374n)return replace(v,{4:replace(definition,{2:set([msId(1027,'definition/task-a'),b])})});

  return v;
 });
 return {registry:list([set([...extended,replace(taskRow,{1:b}),bodyCriterion]),...registry.slice(1)]),content:set([...content,contentB]),get,holder,observer,task};
}
export const MS_CASES=['shared','independent','orphan','orphan-removed','duplicate','aggregate','negative','unavailable','known-zero','capacity-zero','no-task','no-body','execution-blocked'] as const;

export const msPath=(root:bigint,selector:CanonicalValue,field=1n):StatePath=>({rootStateTypeId:root,fieldId:field,selectors:[{kind:'mapKey',key:selector}]});

export const MS_STAGES=[['observe',10,712,713],['freeze',14,713,714],['select',20,714,716],['task',40,716,717],['assess',50,717,719],['raw',51,719,722],['reasons',52,722,726],['decision',60,726,728],['intent',70,728,729],['expression',80,729,730],['plan',90,730,731],['attempt',100,731,732],['outcome',110,732,733],['replenish',110,651,652],['deadline',140,371,372]] as const;
export function multisourceRegistrations(profile:CanonicalValue){
 const p=rec(profile,710n),O=f(p,1n),C=f(p,2n),tasks=items(f(p,3n),'list').map(t=>f(rec(t,708n),1n)),reserves=items(f(p,5n),'list').map(v=>f(rec(v,707n),1n));
 return list(MS_STAGES.map(([name,phase,input,output],index)=>{
  const reads=name==='observe'?[msPath(268n,O),...reserves.map(k=>msPath(649n,k))]:name==='task'?[msPath(268n,O),...tasks.flatMap(t=>[msPath(373n,t),msPath(373n,t,2n)])]:name==='assess'?[msPath(487n,C)]:name==='replenish'?reserves.map(k=>msPath(649n,k)):name==='deadline'?tasks.map(t=>msPath(373n,t)):[];
  const writes=name==='replenish'?reserves.map(k=>msPath(649n,k)):name==='deadline'?tasks.map(t=>msPath(373n,t)):[];
  const fields=[u(index+1),msId(1001,'event/multisource/'+name),u(phase),list([u(input)]),u(output),list(reads.map(p=>statePathPatternValue(msPattern(p)))),list(writes.map(p=>statePathPatternValue(msPattern(p))))];
  if(writes.length)fields.push(msId(1025,name==='replenish'?'authority/general-attention-local-reserve':'authority/prospective-commitments'));
  return r(711,fields);
 }));
}
export function multisourceRecipe(caseName:MultisourceCase,law:typeof MS_LAWS[number]){
 if(!MS_CASES.includes(caseName)||!MS_LAWS.includes(law)||law==='DescriptionDice'&&!['duplicate','aggregate','orphan-removed'].includes(caseName))throw Error('MULTISOURCE_RECIPE_DOMAIN');
 const b=multisourceBase(),{holder:C,observer:O,get}=b,bodyGround=record(rec(old(494,[u(1),b.task('a')]),494n).schema,new Map([[1n,u(2)],[3n,old(493,[C,msDefinition('body-criterion')])]]));
 const tasks=['a','b'].map(n=>r(708,[b.task(n),msId(1027,'definition/task-'+n),msId(1027,'definition/task-instruction-'+(n==='a'?'two':'one')),old(392,[rational(1,4),!(n==='a'&&['orphan','orphan-removed'].includes(caseName))]),replace(get('task-workspace'),{2:u(1),3:true,4:false}),get('task-concern')]));
 const reserve=(n:string)=>ga('LocalReserveKey',[C,msId(1044,'local-reserve/'+n)]);
 const channels=[['a','A',10],...(caseName==='aggregate'?[['b','A',20]]:[]),['otherB','B',10],['otherC','C',10]].map(([ch,n,width])=>ga('LocalReserveChannel',[msId(1005,'channel/'+ch),O,reserve(String(n)),msId(1045,'interoceptive-signal/'+n),rational(Number(width),1),true,caseName!=='unavailable']));
 const descriptions:CanonicalValue[]=[];

 descriptions.push(body('body-a',['a']));if(caseName==='duplicate')descriptions.push(body('body-b',['a']));if(caseName==='aggregate')descriptions.push(body('body-b',['b']),body('body-c',['a','b']));
 if(caseName!=='orphan-removed'&&law!=='DescriptionDice')descriptions.push(r(709,[msDefinition('task-situation'),old(494,[u(1),b.task('a')]),msId(1027,'definition/protocol-contact-two'),msId(1027,'definition/task-instruction-two'),msId(1045,'interoceptive-signal/'+(caseName==='independent'?'B':'A')),list([msId(1005,'channel/'+(caseName==='independent'?'otherB':'a'))]),rational(60,1),u(caseName==='negative'?2:1)]));

 const initial=new AuthoritativeState([{path:msPath(268n,O),value:old(267,[C])},{path:msPath(487n,C),value:old(486,[set(caseName==='no-body'?[]:[msId(1027,'definition/embodied-response-a')])])},...['A','B','C'].map(n=>({path:msPath(649n,reserve(n)),value:old(454,[rational(caseName==='known-zero'?70:20,1),signed(0)])})),...['a','b'].filter(n=>n!=='a'||caseName!=='no-task').flatMap(n=>[{path:msPath(373n,b.task(n)),value:old(372,[u(1)])},{path:msPath(373n,b.task(n),2n),value:old(390,[msId(1027,'definition/task-instruction-'+(n==='a'?'two':'one'))])}])]);
