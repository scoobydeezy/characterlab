import {it,expect} from 'vitest';
import {runIntention as run,intentionSafeView as safe,validateFrames,LAWS,type Law} from '../campaign3/intentionForgetting';
import {intentionCases as cases,make} from './intentionForgettingFixtures';
const intents=(s:ReturnType<typeof run>)=>s.rows.map(r=>(r.expression as {intent:number}).intent);
it('recovers retained but inaccessible prospective content, then actually executes',()=>{const s=run('RetainedCue',cases().main);expect(intents(s)).toEqual([0,0,0,0,1,0,0,0]);expect(s.rows[2].goal).toBe('Open');expect((s.rows[2].retained as unknown[]).length).toBe(2);expect(s.rows[3].cueDisposition).toBe('UnavailableCue');expect(s.world[4].success).toBe(true);expect(s.state.goal).toBe('Fulfilled');});
it('capacity loss defeats exactly the same later reminder without cancelling desire',()=>{const s=run('RetainedCue',cases().lost);expect(intents(s)).toEqual(Array(8).fill(0));expect(s.rows[1].losses).toHaveLength(1);expect(s.rows[4].goal).toBe('Open');expect(s.rows[4].recall).toEqual([2]);expect(s.state.goal).toBe('Expired');});
it('NoLoss rescues recall under pressure; PersistentAccess rescues absent-cue access but not erased content',()=>{expect(intents(run('NoLoss',cases().lost))[4]).toBe(1);expect(intents(run('PersistentAccess',cases().noCue))[2]).toBe(1);expect(intents(run('PersistentAccess',cases().lost))).toEqual(Array(8).fill(0));});
it('retained routeB controls exclude a hidden fixed action reconstruction',()=>{expect(intents(run('RetainedCue',cases().routeB))[4]).toBe(2);expect(intents(run('RetainedCue',cases().lostB))).toEqual(Array(8).fill(0));expect(intents(run('NoLoss',cases().lostB))[4]).toBe(2);});
it('wrong and absent cues differ from forgotten content and never manufacture instructions',()=>{for(const n of ['noCue','wrongCue'] as const){const s=run('RetainedCue',cases()[n]);expect(intents(s)).toEqual(Array(8).fill(0));expect(s.state.memory).toHaveLength(2);}expect(run('RetainedCue',cases().wrongCue).rows[4].recall).toEqual([2]);});
it('cancel, expiry and failed opportunity keep independent causes',()=>{for(const n of ['cancelled','expired','noOpportunity'] as const)expect(intents(run('RetainedCue',cases()[n]))).toEqual(Array(8).fill(0));expect(run('RetainedCue',cases().cancelled).state.goal).toBe('Cancelled');expect(run('RetainedCue',cases().expired).state.goal).toBe('Expired');expect(run('RetainedCue',cases().noOpportunity).rows[4].recall).toEqual([1]);});
it('execution failure preserves earlier expression, and hidden success cannot fulfill a goal',()=>{const a=run('RetainedCue',cases().main),b=run('RetainedCue',cases().blocked);expect(a.rows[4].expression).toEqual(b.rows[4].expression);expect(b.world[4].success).toBe(false);expect(b.state.goal).toBe('Expired');expect(run('RetainedCue',cases().hidden).state.goal).toBe('Expired');});
it('whole later safe views exclude denied instruction and hidden world truth under matched evidence',()=>{for(const [a,b] of [['hidden','hiddenBlocked'],['falseReceipt','trueReceipt'],['denied','deniedB']] as const)expect(safe(run('RetainedCue',cases()[a]))).toEqual(safe(run('RetainedCue',cases()[b])));});
it('retention pressure requires an actual competing acquisition and delayed cue can recover before expiry',()=>{expect(intents(run('RetainedCue',cases().noNeutral))[4]).toBe(1);expect(intents(run('RetainedCue',cases().lateCue))[5]).toBe(1);});
it('reminder source at4 acts only next step and never re-encodes a lost instruction',()=>{const s=run('RetainedCue',cases().lost);expect(s.rows[3].observation).toMatchObject({cue:1,instruction:0});expect(s.state.memory.map(x=>x.id)).toEqual([2]);expect(s.rows[4].losses).toEqual([]);});
for(const law of LAWS)it(`all component prefixes and exact continuations replay under ${law}`,()=>{for(const input of Object.values(cases())){const full=run(law,input);for(let i=0;i<=8;i++){const a=run(law,input,i);expect(a.rows).toEqual(full.rows.slice(0,i));expect(a.world).toEqual(full.world.slice(0,i));if(i)expect(a.state.goal).toBe(a.rows[i-1].goal);}}},60000);
it('rejects malformed schedules, late instructions, out-of-domain capacity and callback properties',()=>{expect(()=>run('RetainedCue',cases().main.slice(1))).toThrow();expect(()=>run('RetainedCue',make(at=>({instruction:at===3?1:0})))).toThrow();expect(()=>run('RetainedCue',make(()=>({capacity:0})))).toThrow();const xs=cases().main;Object.defineProperty(xs[0],'instruction',{get(){throw Error('callback invoked');}});expect(()=>validateFrames(xs)).toThrow('INTENTION_DATA');expect(()=>run('bogus' as Law,cases().main)).toThrow('LAW');});
it('source frames and returned prefixes do not mutate each other',()=>{const input=cases().main,before=structuredClone(input),s=run('RetainedCue',input,4);s.rows.length=0;expect(input).toEqual(before);expect(run('RetainedCue',input).rows).toHaveLength(8);});
