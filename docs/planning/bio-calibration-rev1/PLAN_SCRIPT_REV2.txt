import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {createServer} from 'vite';
const p='docs/planning/',planFile=p+'BIO_COMPARISON_PLAN_REV2.json',outFile=p+'BIO_COMPARISON_RESULT_REV2.json',sha=b=>createHash('sha256').update(b).digest('hex');
const server=await createServer({configFile:false,server:{middlewareMode:true,hmr:false},appType:'custom'});
try{const bio=await server.ssrLoadModule('/src/campaign3/bioComparison.ts'),data=await server.ssrLoadModule('/src/campaign2/canonicalData.ts'),codec=await server.ssrLoadModule('/src/campaign3/receivingCodecs.ts'),canonical=await server.ssrLoadModule('/src/substrate/canonicalEncoding.ts');const items=data.dataItems,f=data.dataField,rec=data.dataRecord,key=data.dataKey;
 if(process.argv.includes('--plan')){
  const calibration=[];for(let seed=0;seed<32;seed++){const run=bio.createBioComparison('Earned',{earlySeed:seed,contrarySeeds:Array(12).fill(0)});for(let i=0;i<5;i++)await run.step();const summaries=items(items(run.snapshot(),'list')[3],'list').map(bio.bioSummary);calibration.push({seed,choices:summaries.slice(0,4).map(x=>x.chosen),evidence:summaries.slice(0,4).map(x=>x.evidence),probe:summaries[4].modifiers,history:summaries[4].history});}
  fs.writeFileSync(p+'BIO_CALIBRATION_REV2.json',JSON.stringify({status:'CALIBRATION ONLY',calibration},null,2)+'\n',{flag:'wx'});
  const first=calibration.find(x=>x.choices.every(c=>c==='A'));assert(first,'four positive choices witness');const second=calibration.find(x=>JSON.stringify(x.probe)!==JSON.stringify(first.probe));assert(second,'different earned modifier witness');
  const attempts=[],contrarySeeds=[];for(let at=6;at<=17;at++){let found=false;for(let seed=0;seed<254;seed++){const row=await bio.bioStep(BigInt(at),seed,codec.receivingRecord(414,[canonical.list([])]),'Earned'),q=rec(f(rec(row.qualification,429n),3n),430n),e=data.dataUnsigned(f(q,1n))===1n?f(q,3n):undefined;attempts.push({at,seed,evidence:e?`${e.numerator}/${e.denominator}`:'None'});if(e&&e.numerator<0n){contrarySeeds.push(seed);found=true;break;}}assert(found,'contrary seed witness '+at);}
  const paths=['docs/formal/BIO_COMPARISON_CONTRACT.md','src/campaign3/bioComparison.ts','src/campaign3/longitudinalMath.ts','src/campaign3/longitudinalModel.ts','src/campaign2/cognitiveMath.ts','src/campaign2/cognitiveTransforms.ts','src/campaign2/cognitiveArbitration.ts','src/campaign2/cognitiveChoice.ts','src/substrate/random.ts','scripts/qualify-bio-comparison.mjs'];
  fs.writeFileSync(planFile,JSON.stringify({status:'FROZEN after declared calibration, before qualification matrix',scope:'Composed component experiment; no new public scheduler/factory qualification.',first:first.seed,second:second.seed,contrarySeeds,probeSeed:255,nondecisionSeed:254,calibration,contraryAttempts:attempts,artifacts:paths.map(path=>({path,sha256:sha(fs.readFileSync(path))})),cases:[...bio.BIO_LAWS.flatMap(law=>[first.seed,second.seed].map(earlySeed=>({law,earlySeed,name:law+'/'+earlySeed}))),{law:'Earned',earlySeed:first.seed,name:'coupled-repeat'}]},null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({first:first.seed,second:second.seed,contrarySeeds,calibrations:calibration.length,contraryAttempts:attempts.length}));
 }else{assert(!fs.existsSync(outFile));const plan=JSON.parse(fs.readFileSync(planFile));for(const a of plan.artifacts)assert.equal(sha(fs.readFileSync(a.path)),a.sha256,a.path);const results=[];
  try{for(const c of plan.cases){const input={earlySeed:c.earlySeed,contrarySeeds:plan.contrarySeeds},run=bio.createBioComparison(c.law,input),saves=[run.save()];while(await run.step())saves.push(run.save());for(let i=0;i<saves.length;i++){const restored=await bio.restoreBioComparison(c.law,input,saves[i]);assert.deepEqual(restored.save(),saves[i]);assert.equal(await restored.step(),i+1<saves.length);assert.deepEqual(restored.save(),saves[Math.min(i+1,saves.length-1)]);}results.push({...c,prefixes:saves.length,saveSha256:sha(run.save()),rows:items(items(run.snapshot(),'list')[3],'list').map(bio.bioSummary)});console.log(c.name+': '+saves.length+' component prefixes');}
   const get=(law,seed=plan.first)=>results.find(r=>r.law===law&&r.earlySeed===seed),a=get('Earned'),b=get('Earned',plan.second),probe=r=>r.rows[4],end=r=>r.rows[17],sign=s=>Number(s.split('/')[0]);
   assert.equal(a.saveSha256,results.at(-1).saveSha256);assert.equal(a.rows[0].context,b.rows[0].context);assert.equal(a.rows[0].raw,b.rows[0].raw);assert.equal(a.rows[0].reasons,b.rows[0].reasons);assert.notEqual(probe(a).history,probe(b).history);assert.notDeepEqual(probe(a).modifiers,probe(b).modifiers);assert.notDeepEqual(probe(a).probabilities,probe(b).probabilities);
   for(const r of results){assert.deepEqual(r.rows.map(x=>x.nondecision),a.rows.map(x=>x.nondecision));for(const i of [4,17,18])assert.equal(r.rows[i].seed,255);for(const x of r.rows.slice(5,17)){assert(sign(x.evidence)<0);const [n,d]=x.authorship.split('/').map(BigInt);assert(n*2n>=d);}}
   assert(sign(a.rows[3].strength)>0);assert(sign(a.rows[5].strength)>0);assert(sign(a.rows[16].strength)<0);assert.notDeepEqual(probe(a).modifiers,end(a).modifiers);
   for(const seed of [plan.first,plan.second]){const base=get('Earned',seed);for(const law of ['DisplayOnly','HistoryOnly']){const c=get(law,seed);assert.equal(probe(c).history,probe(base).history);assert(probe(c).modifiers.every(x=>x==='0'));}assert.equal(get('Refold',seed).rows[18].history,base.rows[18].history);assert.deepEqual(get('Refold',seed).rows,base.rows);assert.notEqual(get('RecomputedHistory',seed).rows[18].archive,base.rows[18].archive);assert.equal(base.rows[18].archive,base.rows[16].archive);}
   assert.deepEqual(probe(get('HistoryOnly')).probabilities,probe(get('HistoryOnly',plan.second)).probabilities);assert.deepEqual(probe(get('AuthoredTrait')).probabilities,probe(get('AuthoredTrait',plan.second)).probabilities);assert.notEqual(probe(get('DisplayOnly')).display,probe(get('HistoryOnly')).display);
   fs.writeFileSync(outFile,JSON.stringify({status:'PASS',scope:plan.scope,planSha256:sha(fs.readFileSync(planFile)),runs:results.length,candidates:6,prefixes:results.reduce((n,r)=>n+r.prefixes,0),advancing:results.reduce((n,r)=>n+r.prefixes-1,0),terminal:results.length,results},null,2)+'\n',{flag:'wx'});console.log('PASS BIO matrix');
  }catch(error){fs.writeFileSync(p+'BIO_COMPARISON_FAILED_REV2.json',JSON.stringify({status:'FAIL',error:String(error),planSha256:sha(fs.readFileSync(planFile)),completed:results},null,2)+'\n',{flag:'wx'});throw error;}}
}finally{await server.close();}
